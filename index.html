{% highlight kotlin %} class MyGame(context: Context) : ContextListener(context) { override suspend fun Context.start() { val logo = resourcesVfs["logo.png"].readTexture() val format = graphics.preferredFormat val batch = SpriteBatch(graphics.device, graphics, format) val camera = OrthographicCamera(960, 540) val clear = Color.BLACK onUpdate { renderFrame { frame, encoder -> val pass = encoder.beginRenderPass( RenderPassDescriptor(colorAttachment(frame, clear)) ) camera.update() batch.begin() batch.draw(logo, 50f, 50f) batch.flush(pass, camera.viewProjection) batch.end() pass.end() } } } } {% endhighlight %}

{% highlight kotlin %} val atlas = resourcesVfs["tiles.atlas.json"].readAtlas() val hero = atlas["hero"].slice onUpdate { dt -> renderFrame { frame, encoder -> val pass = encoder.beginRenderPass( RenderPassDescriptor(colorAttachment(frame, Color.BLACK)) ) batch.begin() batch.draw(hero, x, y, rotation = angle) batch.flush(pass, camera.viewProjection) batch.end() pass.end() } } {% endhighlight %}
{% highlight kotlin %} val fox = resourcesVfs["models/Fox.glb"].readGltfModel( config = GltfLoaderPbrConfig(GltfModelConfig(skinned = true)) ) fox.enableAnimation(1) val environment = PBREnvironment(device).apply { val sun = DirectionalLight(direction = Vec3f(0.3f, 0.5f, 0.8f)) setDirectionalLight(sun) } onUpdate { dt -> fox.update(dt) renderFrame { frame, encoder -> val pass = encoder.beginRenderPass( RenderPassDescriptor( colorAttachment(frame, Color.DARK_GRAY), depthStencilAttachment = depthTarget.depthAttachment(), ) ) modelBatch.render(fox, camera, environment) modelBatch.flush(pass, camera, dt) pass.end() } } {% endhighlight %}

Why LittleKt

A framework, not an engine

Inspired by libGDX and KorGE. You stay in control of your game loop, your code and your architecture.

Free forever

Apache 2.0 licensed. No fees, no royalties and no splash screens. What you build is yours.

Build your own engine

Use only what you need. Every layer is open, from the scene graph down to the GPU device.

One codebase

Share your game code across desktop and web with Kotlin Multiplatform.
